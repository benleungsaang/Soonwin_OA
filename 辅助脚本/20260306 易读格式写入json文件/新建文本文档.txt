with open('./123.json', 'r' , encoding='utf-8') as f:
    data = json.loads(f.read())

with open('./123.json', mode='w', encoding='utf-8') as f:
    json.dump(
        data,
        f,
        ensure_ascii=False,  # 中文不转码
        indent=4,            # 4个空格缩进，层级清晰
        separators=(',', ': ')  # 键值间加空格，更易读（默认就是这个，可省略）
    )
